# 多模态系统依赖
#
# 文档 §8：每台机器在仓库根目录 `pip install -r requirements.txt`。
# 公共依赖默认启用；各节点的额外依赖以注释分区列出，**由各人按需自行启用**，
# 避免 B/C 的机器被迫装上 vllm 这种巨物。
#
# 版本按 2026-09 的实测环境锁定。下面几处约束是踩过坑才定下来的，
# 不是随手写的，改之前先看注释。

# ── 公共（三台机器都要）──────────────────────────────────────
# fastapi 用范围而不是精确锁：vllm 要求 fastapi>=0.133,<0.137，锁死 0.141 会让
# A 的机器解析失败。用范围后 A 自动取 0.136.x，B/C 仍取 0.141.x——
# 三个节点只通过 HTTP 通信，版本略有差异不影响。
fastapi>=0.133,<0.142
uvicorn[standard]==0.53.0
pydantic==2.13.5
numpy==2.2.6
opencv-python-headless==5.0.0.93
pillow==12.3.0
httpx==0.28.1          # common/selftest.py 的 TestClient 需要

# ── 深度学习公共（三台机器都装同一套）────────────────────────
# 这三个版本是三个节点能共用的关键。vllm 0.27.0 起把 torch 跳到了 2.13.0，
# 会和视觉节点分家，所以 A 那边必须用 vllm 0.26.0（见下方 A 区）。
# 视觉节点的组合（torch 2.11 + ultralytics + transformers）已实测跑通。
torch==2.11.0
torchvision==0.26.0

# ── A · llm_node（16GB 节点）──────────────────────────────────
# 网关 + LangChain Agent + vLLM。
#
# ⚠️ vllm 必须用 0.26.0，不要升到 0.27+：
#    0.26.0 → torch==2.11.0 / torchvision==0.26.0 / flashinfer 0.6.14
#    0.27.0 → torch==2.13.0 / torchvision==0.28.0  ← 与上面公共区冲突
#
# ⚠️ 系统需装 CUDA toolkit >= 12.9（仅 A 的机器需要）：
#    vllm 依赖的 flashinfer 会为 sm_120 现场编译 kernel，需要 nvcc。系统若只有
#    CUDA 12.8，flashinfer 的版本检查会直接失败。已在 CUDA 13.0 上实测通过。
#    （只装 cuda-toolkit-13-0，**不要**装 cuda 元包，那会连带换掉驱动。）
#
# vllm 只装在能跑 Linux 的环境（16GB 目标机 / WSL2），Windows 侧不装：
#   pip install vllm==0.26.0
# 网关侧（Windows）只装下面三个编排依赖，2026-09-20 由 A 启用：
langchain>=1.0
langchain-openai>=1.0
openai

# ── B · vision_fast（8GB #1 节点）─────────────────────────────
# YOLOv8s + ResNet50，模型常驻显存。torch/torchvision 用上面的公共区。
# ultralytics

# ── C · vision_heavy（8GB #2 节点）────────────────────────────
# ocr + stylize，懒加载。torch/torchvision 用上面的公共区。
#
# OCR 走 transformers 加载 PaddleOCR-VL（0.9B）。权重约 1.8GB 需另放本机，
# 见 vision_heavy/README.md；或用 OCR_MODEL_ID 指定别的权重。
# 若不使用 PaddleOCR-VL，可改用 PP-OCRv6（34.5M，纯 CPU 可跑，几乎不占显存），
# transformers 同样内置支持，换 OCR_MODEL_ID 即可，无需额外依赖。
transformers==5.17.0
